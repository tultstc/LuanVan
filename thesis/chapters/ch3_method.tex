\chapter{Hybrid FGTN-Mamba: phương pháp đề xuất}
\label{ch:method}

Chương này mô tả khối Hybrid FGTN-Mamba và pipeline phát hiện hoàn chỉnh. Mọi mô tả đều bám theo cài đặt trong thư mục \path{projects/fgtn_ava} của kho mã và các siêu tham số nêu dưới đây là giá trị thực tế trong cấu hình huấn luyện.

\section{Tổng quan kiến trúc}
\label{sec:overview}

Pipeline đề xuất (\cref{fig:pipeline}) gồm năm bước:
\begin{enumerate}
  \item backbone Video Swin-Tiny \cite{liu2022videoswin} đã tiền huấn luyện trên Kinetics-400 trích đồng thời bản đồ đặc trưng của tầng 3 và tầng 4;
  \item mỗi tầng đi qua một khối Hybrid FGTN-Mamba riêng với tham số phù hợp độ phân giải và mức ngữ nghĩa của tầng đó, nhờ vậy đặc trưng được làm giàu theo thời gian trước khi cắt theo vùng đề xuất người (person proposal);
  \item 3D RoIAlign lấy đặc trưng của từng person proposal ở cả hai tỉ lệ rồi nối theo kênh (multi-scale RoI), trong đó tầng 3 giữ các chi tiết như điện thoại, bàn tay, bàn chân còn tầng 4 giữ ngữ nghĩa toàn thân;
  \item ACRN học quan hệ giữa từng actor với bàn, ghế, thiết bị và cảnh xung quanh;
  \item head sigmoid dự đoán xác suất độc lập cho từng nhãn và được huấn luyện bằng focal loss kết hợp trọng số lớp. Khi suy luận, ngưỡng theo lớp và logic theo track giúp giảm cảnh báo chớp tắt.
\end{enumerate}

Thành phần mới đóng vai trò trung tâm là Hybrid FGTN-Mamba (bước 2) còn đặc trưng đa tỉ lệ, ACRN và hàm mất mát hai cấp là các thành phần đã có tiền lệ, được tích hợp vào để xử lý các hạn chế còn lại (\cref{sec:multiscale,sec:acrn,sec:loss}). Các tầng của Video Swin-T được đánh số từ 1 đến 4 như trong \cref{fig:architecture}, trong đó tầng 3 và tầng 4 tương ứng với \code{out\_indices} 2 và 3 trong mã nguồn (``stage 2'' và ``stage 3'' khi đánh số từ 0). Với đoạn video $V \in \R^{3 \times 32 \times 256 \times 256}$ và $N$ hộp người, chuỗi biến đổi của pipeline đầy đủ là
\begin{equation}
\begin{aligned}
V &\xrightarrow{\text{Swin-T}} \big(F^{(3)}, F^{(4)}\big)
  \xrightarrow{\text{HFM}_3,\ \text{HFM}_4} \big(\tilde{F}^{(3)}, \tilde{F}^{(4)}\big)\\
  &\xrightarrow{\text{RoIAlign}} r_k
  \xrightarrow{\text{ACRN}} c_k
  \xrightarrow{\text{head}} \mat{p}_k \in [0,1]^{K},
\end{aligned}
\label{eq:flow}
\end{equation}
trong đó $F^{(3)} \in \R^{384 \times 16 \times 16 \times 16}$, $F^{(4)} \in \R^{768 \times 16 \times 8 \times 8}$, $\text{HFM}_s$ là ngăn xếp hai khối Hybrid FGTN-Mamba của tầng $s$, $r_k \in \R^{1152 \times 1 \times 8 \times 8}$ là đặc trưng RoI đa tỉ lệ và $c_k$ là đầu ra của ACRN. Biến thể chỉ dùng tầng 4 (cấu hình A6 trong \cref{tab:ablation}) không dùng tầng 3 và ACRN nên chuỗi biến đổi rút gọn thành $V \rightarrow F^{(4)} \rightarrow \tilde{F}^{(4)} \rightarrow r_k \in \R^{768 \times 1 \times 8 \times 8} \rightarrow \mat{p}_k$.

\section{Vị trí đặt khối: hậu-backbone, tiền-RoI}
\label{sec:placement}

Khối được đặt sau backbone và trước 3D RoIAlign vì ba lý do. Thứ nhất, nếu đặt bên trong Video Swin thì phải sửa kiến trúc đã tiền huấn luyện và trọng số Kinetics-400 sẽ không còn nạp được nguyên vẹn. Thứ hai, nếu đặt sau RoIAlign thì mô hình thời gian chỉ nhìn thấy vùng người đã bị cắt và mất tín hiệu cảnh, hơn nữa \code{SingleRoIExtractor3D} gộp trung bình theo thời gian trước khi cắt nên sau RoI không còn trục $T$. Thứ ba, khi đặt tiền-RoI, mỗi vị trí không gian được làm giàu bằng lịch sử dọc trục thời gian rồi RoIAlign mới lấy phần liên quan đến từng actor và giao diện ``tensor vào, tensor cùng kích thước ra'' cho phép thay backbone mà không phải đổi neck. Vị trí này cũng xác định phạm vi cải tiến của khối, tức là khối chỉ cải thiện biểu diễn theo thời gian, không sửa lỗi của person proposal và không tự giải quyết được vấn đề mất cân bằng dữ liệu.

\section{Đầu vào, định dạng lại và chuẩn hóa}
\label{sec:reshape}

Với bản đồ đặc trưng $F \in \R^{B \times C \times T \times H \times W}$, neck chỉ mô hình hóa dọc trục $T$ và không trộn thông tin không gian thêm lần nữa. Tensor được hoán vị và định dạng lại thành
\begin{equation}
\mat{X} = \mathrm{reshape}\big(\mathrm{permute}(F)\big) \in \R^{(B \cdot H \cdot W) \times C \times T},
\label{eq:reshape}
\end{equation}
tức là mỗi tọa độ không gian trở thành một chuỗi độc lập gồm $T$ token $C$ chiều. Sau khi qua các khối, tensor được đưa về dạng ban đầu. Mỗi khối dùng chuẩn hóa trước (pre-norm) trên chiều kênh tại từng bước thời gian:
\begin{equation}
\bar{\mat{X}} = \LN(\mat{X}), \qquad \epsilon = 10^{-6}.
\label{eq:prenorm}
\end{equation}
Ở tầng 4, ta có $C = 768$, $T = 16$ (32 khung hình qua patch embedding thời gian với bước 2), $H = W = 8$ và số chuỗi là $B \cdot 64$. Ở tầng 3, ta có $C = 384$, $H = W = 16$ và số chuỗi là $B \cdot 256$.

\section{Nhánh toàn cục: mô hình không gian trạng thái chọn lọc hai chiều}
\label{sec:ssm-branch}

Nhánh SSM nhắm vào hai hạn chế L1 và L2 (\cref{tab:gaps}). Với những hành vi bền vững như ngủ trên bàn hay gác chân, vài khung hình quanh khung hình khóa không đủ để phân biệt một tư thế tạm thời với một hành vi kéo dài, trong khi quét chọn lọc (selective scan) truyền một trạng thái qua toàn bộ chuỗi với số bước tuyến tính theo $T$. Nhánh này dùng khối Mamba \cite{gu2023mamba} theo cài đặt \code{bimamba v2} của Video Mamba Suite \cite{chen2024videomambasuite} với $N = 16$, hệ số mở rộng $E = 2$ ($C' = 2C$), hạng phép chiếu bước thời gian $\lceil C/16 \rceil$ và tích chập nhân quả theo kênh với $k = 4$. Trên mỗi chuỗi $\bar{\mat{X}} \in \R^{T \times C}$, các phép biến đổi là
\begin{align}
[\mat{U};\ \mat{Z}] &= \bar{\mat{X}}\mat{W}_{\text{in}}, \qquad \mat{U}, \mat{Z} \in \R^{T \times C'}, \label{eq:mamba-in}\\
\mat{U}' &= \mathrm{SiLU}\big(\mathrm{DWConv1d}_{k=4}(\mat{U})\big), \label{eq:mamba-conv}\\
\Delta_t &= \mathrm{softplus}\big(\mat{W}_\Delta \mat{W}_{\text{dt}} \mat{u}'_t + \mat{b}_\Delta\big), \quad
\mat{B}_t = \mat{W}_B \mat{u}'_t, \quad \mat{C}_t = \mat{W}_C \mat{u}'_t, \label{eq:mamba-sel}\\
\mat{h}_t &= \exp(\Delta_t \mat{A})\,\mat{h}_{t-1} + \Delta_t \mat{B}_t \mat{u}'_t, \qquad
y_t = \mat{C}_t^{\top}\mat{h}_t + D\,u'_t, \label{eq:mamba-scan}\\
\mat{S}^{\rightarrow} &= \mat{Y} \odot \mathrm{SiLU}(\mat{Z}). \label{eq:mamba-gate}
\end{align}
Ở đây, $\mat{A} = -\exp(\mat{A}_{\log})$ là tham số học được còn $\Delta_t$, $\mat{B}_t$, $\mat{C}_t$ được sinh từ đầu vào tại mỗi bước nên mức giữ hay quên trạng thái thay đổi theo nội dung. Đoạn gần như tĩnh sẽ duy trì trạng thái lâu hơn còn thay đổi đột ngột làm tăng trọng số cho đầu vào mới. Tính chọn lọc này quan trọng hơn việc chỉ thay attention bằng một phép hồi quy và phù hợp với dữ liệu có cả hành vi bền vững lẫn thao tác nhanh. Do đoạn video được lấy đối xứng quanh khung hình khóa, \code{bimamba v2} dùng thêm một bộ tham số quét riêng cho chiều ngược nhưng dùng chung $\mat{W}_{\text{in}}$ và $\mat{W}_{\text{out}}$:
\begin{equation}
\mat{S}^{\leftarrow} = \mathrm{flip}\big(\Scan_b(\mathrm{flip}(\bar{\mat{X}}))\big), \qquad
\mat{S} = \tfrac{1}{2}\big(\mat{S}^{\rightarrow} + \mat{S}^{\leftarrow}\big)\mat{W}_{\text{out}} \in \R^{T \times C}.
\label{eq:bimamba}
\end{equation}
Quét ngược bổ sung thông tin từ các khung hình sau khung hình khóa nhưng đổi lại, hệ cảnh báo trực tuyến phải chịu một độ trễ nhìn trước (look-ahead) tương ứng (\cref{sec:inference}).

\section{Nhánh cục bộ: self-attention trong cửa sổ thời gian}
\label{sec:local-branch}

Do SSM nén lịch sử vào trạng thái nên nó có thể làm mờ những chuyển tiếp rất ngắn như pha đưa điện thoại lên hay lúc bắt đầu một thao tác nguy hiểm. Nhánh cục bộ được thêm vào để giữ đường so khớp trực tiếp giữa mỗi khung hình với các khung hình lân cận. Từ $\bar{\mat{X}}$, phép chiếu $\mat{W}_{\text{qkv}} \in \R^{C \times 3C}$ tạo ra $\mat{Q}, \mat{K}, \mat{V}$ được chia thành $h = 4$ đầu. Với $W = 5$, token $i$ chỉ chú ý tới các token $j$ thỏa mãn $|i - j| \leq 2$:
\begin{equation}
\mat{A}_{\text{loc}} = \softmax\!\left(\frac{\mat{Q}\mat{K}^{\top}}{\sqrt{d}} + \mat{M}\right)\mat{V}\,\mat{W}_{\text{proj}}, \qquad
M_{ij} = \begin{cases} 0, & |i - j| \leq 2,\\ -\infty, & \text{ngược lại.}\end{cases}
\label{eq:local-attn}
\end{equation}
Ở biên đoạn video, mặt nạ chỉ giữ các phần tử hợp lệ chứ không quấn vòng còn bốn đầu attention cho phép các không gian con kênh học những kiểu chuyển động khác nhau. Vì tại neck $T = 16$ nên ma trận $T \times T$ được tính đầy đủ rồi mới che với chi phí không đáng kể. Khi $T$ lớn, cùng công thức này có thể được thực thi theo dạng dải với chi phí $O(TWC)$ \cite{beltagy2020longformer}. Nhánh cục bộ hiện chưa dùng mã hóa vị trí tương đối và thông tin thứ tự thời gian được nhánh SSM cung cấp gián tiếp qua đường residual.

\section{Cổng sigmoid: định tuyến dài hạn--ngắn hạn theo nội dung}
\label{sec:gate}

Nếu cộng $\mat{S}$ và $\mat{A}_{\text{loc}}$ với trọng số cố định thì mọi kênh và mọi đoạn video đều nhận cùng một tỷ lệ dài hạn--ngắn hạn, điều này mâu thuẫn với đặc thù của các hành vi đã nêu ở \cref{sec:behaviours}. Vì vậy, luận văn dùng cổng sigmoid làm bộ định tuyến mềm, được tính từ đầu vào đã chuẩn hóa bằng tích chập $1 \times 1$ theo kênh:
\begin{equation}
\mat{G} = \sigma\big(\mat{W}_g \bar{\mat{X}} + \mat{b}_g\big) \in (0,1)^{C \times T}, \qquad
\mat{Z} = \mat{G} \odot \mat{S} + (\mat{1} - \mat{G}) \odot \mat{A}_{\text{loc}}.
\label{eq:gate}
\end{equation}
Cổng phụ thuộc vào nội dung, vị trí không gian, kênh và bước thời gian nên cùng một kênh có thể ưu tiên trạng thái dài hạn ở pha tĩnh rồi chuyển sang nhánh cục bộ tại thời điểm chuyển tiếp. Do cổng không chọn cứng một nhánh nên gradient vẫn truyền được qua cả hai nhánh. Đầu ra của khối là
\begin{equation}
\mat{Y} = \mat{X} + \mathrm{DropPath}_{p = 0{,}1}(\mat{Z}),
\label{eq:residual}
\end{equation}
trong đó DropPath \cite{huang2016droppath} loại bỏ ngẫu nhiên toàn bộ nhánh dư của một chuỗi trong lúc huấn luyện. Neck gồm hai khối như vậy xếp chồng lên nhau.

Cùng một module cài đặt bốn chế độ thông qua tham số \code{branch\_mode} để phục vụ phần lớn ma trận ablation (\cref{tab:ablation}), cụ thể là \code{hybrid} theo \cref{eq:gate} (A6); \code{no\_gate} với $\mat{Z} = 0{,}5\,\mat{S} + 0{,}5\,\mat{A}_{\text{loc}}$ (A5); \code{mamba\_only} với $\mat{Z} = \mat{S}$ (A3); \code{attn\_only} với $\mat{Z} = \mat{A}_{\text{loc}}$ (A4). Temporal MHSA toàn cục (A1) là trường hợp riêng của \code{attn\_only} khi cửa sổ phủ toàn đoạn (\code{window\_size} $\geq 31$ với $T = 16$), baseline không có neck (A0) tương ứng với \code{model.neck = None} còn SSM một chiều (A2) dùng \code{bimamba\_type = "none"} của \code{mamba-ssm} và hiện chưa có khóa cấu hình tương ứng.

\Cref{alg:block} tóm tắt lan truyền thuận của neck tại một tầng đặc trưng ở chế độ \code{hybrid} theo đúng thứ tự phép tính trong lớp \code{FGTNNeckAVA} và \code{HybridFGTNBlock}. Trong thuật toán, chiều cao và chiều rộng của bản đồ đặc trưng được ký hiệu $H_s$, $W_s$ để không nhầm với kích thước cửa sổ $W$ còn đầu ra trộn của hai nhánh được ký hiệu $\mat{O}$ (chính là $\mat{Z}$ trong \cref{eq:gate}) để không trùng với nhánh cổng $\mat{Z}$ bên trong Mamba. Ba điểm của cài đặt cần lưu ý khi đọc thuật toán. Thứ nhất, hai nhánh và cổng cùng đọc một đầu vào đã chuẩn hóa $\bar{\mat{X}}$ nên chúng độc lập với nhau và có thể tính song song, trong đó cổng được tính từ $\bar{\mat{X}}$ chứ không từ đầu ra của các nhánh. Thứ hai, phép chiếu $\mat{W}_{\text{in}}$ được áp dụng một lần cho mỗi token rồi kết quả được lật theo thời gian để đưa vào bộ quét ngược nên hai chiều quét dùng chung phép chiếu vào và phép chiếu ra nhưng có tích chập, phép chiếu chọn lọc, $\mat{A}_{\log}$ và $D$ riêng. Thứ ba, DropPath lấy một biến Bernoulli cho mỗi chuỗi, tức là cho mỗi vị trí không gian của mỗi đoạn video, rồi chia cho xác suất giữ $1-p$ để kỳ vọng của nhánh dư không đổi. Khi suy luận, bước này là phép đồng nhất nên đầu ra của khối là hàm tất định của đầu vào. Mặt nạ dải được cài đặt dưới dạng mặt nạ logic đưa vào \code{scaled\_dot\_product\_attention} của PyTorch, tương đương với việc cộng $0$ hoặc $-\infty$ như trong \cref{eq:local-attn}.

\begin{algorithm}[htbp]
\caption{Lan truyền thuận của neck Hybrid FGTN-Mamba tại một tầng đặc trưng (chế độ \code{hybrid})}
\label{alg:block}
\begin{algorithmic}[1]
\Require bản đồ đặc trưng $F \in \R^{B \times C \times T \times H_s \times W_s}$; số khối $L = 2$; cửa sổ $W = 5$; số đầu $h = 4$; tỉ lệ DropPath $p = 0{,}1$
\Ensure bản đồ $\tilde{F}$ cùng kích thước với $F$
\State $\mat{X} \gets \mathrm{reshape}\big(\mathrm{permute}(F)\big) \in \R^{(B H_s W_s) \times C \times T}$ \Comment{\cref{eq:reshape}}
\For{$\ell = 1, \dots, L$}
  \State $\bar{\mat{X}} \gets \LN(\mat{X})$ trên chiều kênh tại từng bước, $\epsilon = 10^{-6}$
  \State $[\mat{U};\ \mat{Z}] \gets \bar{\mat{X}}\mat{W}_{\text{in}}$ \Comment{chung cho hai chiều}
  \State $\mat{S}^{\rightarrow} \gets \Scan_f(\mat{U}, \mat{Z})$ \Comment{\crefrange{eq:mamba-conv}{eq:mamba-gate}}
  \State $\mat{S}^{\leftarrow} \gets \mathrm{flip}\big(\Scan_b(\mathrm{flip}(\mat{U}), \mathrm{flip}(\mat{Z}))\big)$
  \State $\mat{S} \gets \tfrac{1}{2}\big(\mat{S}^{\rightarrow} + \mat{S}^{\leftarrow}\big)\mat{W}_{\text{out}}$ \Comment{\cref{eq:bimamba}}
  \State $\mat{Q}, \mat{K}, \mat{V} \gets$ tách $\bar{\mat{X}}\mat{W}_{\text{qkv}}$ thành $h$ đầu, mỗi đầu $d = C/h$ chiều
  \State $M_{ij} \gets 0$ khi $|i - j| \leq \lfloor W/2 \rfloor$; ngược lại $M_{ij} \gets -\infty$
  \State $\mat{A}_{\text{loc}} \gets \softmax\big(\mat{Q}\mat{K}^{\top}/\sqrt{d} + \mat{M}\big)\mat{V}\,\mat{W}_{\text{proj}}$ \Comment{\cref{eq:local-attn}}
  \State $\mat{G} \gets \sigma\big(\mathrm{Conv1d}_{1 \times 1}(\bar{\mat{X}})\big)$ \Comment{\cref{eq:gate}}
  \State $\mat{O} \gets \mat{G} \odot \mat{S} + (\mat{1} - \mat{G}) \odot \mat{A}_{\text{loc}}$
  \If{đang huấn luyện}
    \State lấy $\xi_i \sim \mathrm{Bernoulli}(1 - p)$ cho mỗi chuỗi $i$ và đặt $\mat{O}_i \gets \xi_i \mat{O}_i / (1 - p)$
  \EndIf
  \State $\mat{X} \gets \mat{X} + \mat{O}$ \Comment{\cref{eq:residual}}
\EndFor
\State \Return $\tilde{F} \gets \mathrm{permute}^{-1}\big(\mathrm{reshape}^{-1}(\mat{X})\big)$
\end{algorithmic}
\end{algorithm}

\section{Ổn định huấn luyện và cài đặt dự phòng}
\label{sec:stability}

Pre-norm và residual cho phép khối học phần dư thay vì phải tái tạo toàn bộ bản đồ đặc trưng. Ở thời điểm khởi tạo, hai nhánh chỉ đóng góp nhỏ và pipeline xấp xỉ pipeline không có neck nên trọng số tiền huấn luyện của backbone không bị phá vỡ. Gradient được cắt với chuẩn tối đa 40 theo công thức gốc của MMAction2 cho AVA. Kernel selective scan CUDA của \code{mamba-ssm} được khóa phiên bản vì sai khác giữa các kernel có thể ảnh hưởng tới tốc độ và khả năng tái lập. Khi thư viện \code{mamba-ssm} không khả dụng (ví dụ lúc kiểm thử trên CPU), mã nguồn cung cấp bản cài đặt thuần PyTorch \code{NaiveBiMamba} thực hiện đúng phép hồi quy \cref{eq:mamba-scan} bằng vòng lặp theo $T$ với hai bộ tham số xuôi và ngược. Bản này tương đương về mặt số học nhưng chậm hơn khoảng hai bậc nên chỉ được dùng để kiểm tra kích thước tensor.

\section{Độ phức tạp và so sánh với các temporal neck khác}

\subsection{Đếm chi tiết số phép tính và bộ nhớ của một khối}
\label{sec:complexity-detail}

Để làm rõ các bậc độ phức tạp ở trên, phần này đếm số phép nhân--cộng (multiply-accumulate, MAC) của một khối trên \emph{một chuỗi} độ dài $T$. Số FLOP xấp xỉ gấp đôi số MAC còn chi phí của cả tầng bằng số MAC của một chuỗi nhân với số chuỗi $B \cdot H \cdot W$ và số khối. Ký hiệu $R = \lceil C/16 \rceil$ là hạng phép chiếu bước thời gian, $k = 4$ là kích thước nhân tích chập theo kênh, $h$ là số đầu attention và $w = \lfloor W/2 \rfloor$ là nửa cửa sổ. Các phép theo từng phần tử như LayerNorm, SiLU, sigmoid, softmax, phép trộn của cổng và phép cộng residual chỉ tốn $O(TC)$ hoặc $O(TEC)$ nên được bỏ qua trong phép đếm.

Phần lớn chi phí nằm ở các phép chiếu tuyến tính áp dụng độc lập cho từng token. Phép chiếu vào $\mat{W}_{\text{in}}$ tốn $2E\,TC^2$ MAC, phép chiếu ra $\mat{W}_{\text{out}}$ tốn $E\,TC^2$, phép chiếu $\mat{W}_{\text{qkv}}$ và $\mat{W}_{\text{proj}}$ của nhánh cục bộ lần lượt tốn $3TC^2$ và $TC^2$ còn tích chập $1 \times 1$ của cổng tốn $TC^2$, do đó
\begin{equation}
\mathrm{MAC}_{\text{lin}} = (3E + 5)\,TC^2 \;\overset{E = 2}{=}\; 11\,TC^2.
\label{eq:mac-lin}
\end{equation}
Hệ số 11 khớp với số tham số trong \cref{tab:params} vì mỗi trọng số của một phép chiếu được dùng đúng một lần cho mỗi token. Tổng quát, số tham số của một khối là
\begin{equation}
P_{\text{khối}}(C) = (3E + 5)\,C^2 + 2EC\,(k + 2R + 3N + 3) + 7C,
\label{eq:params-block}
\end{equation}
trong đó số hạng thứ hai là hai bộ tham số quét (mỗi chiều gồm tích chập $EC(k+1)$, \code{x\_proj} $EC(R + 2N)$, \code{dt\_proj} $EC(R+1)$, $\mat{A}_{\log}$ có $EC \cdot N$ và $D$ có $EC$ phần tử) còn số hạng cuối gồm LayerNorm và các độ lệch của nhánh cục bộ và cổng. Với $C = 768$, \cref{eq:params-block} cho đúng 6.957.312 tham số như trong \cref{tab:params}. Với $C = 384$ ($R = 24$), một khối có 1.782.912 tham số và ngăn xếp hai khối của tầng 3 có 3.565.824 tham số, khớp với con số xấp xỉ 3,57 triệu đã nêu.

Phần riêng của nhánh SSM theo mỗi chiều quét gồm tích chập theo kênh ($T \cdot EC \cdot k$), phép chiếu sinh $\Delta_t$, $\mat{B}_t$, $\mat{C}_t$ ($T \cdot EC \cdot (R + 2N)$), phép chiếu $\mat{W}_\Delta$ ($T \cdot R \cdot EC$) và bản thân phép quét. Mỗi bước quét cập nhật trạng thái $\mat{h}_t \in \R^{EC \times N}$ bằng một số hữu hạn phép nhân theo phần tử (tính $\exp(\Delta_t \mat{A})$, $\Delta_t \mat{B}_t u'_t$, phép cập nhật và tích $\mat{C}_t^{\top}\mat{h}_t$) nên chi phí quét là $O(T \cdot EC \cdot N)$ với hằng số nhỏ phụ thuộc vào kernel. Cộng hai chiều lại, ta có
\begin{equation}
\mathrm{MAC}_{\text{ssm}} = 2\,T \cdot EC\,(k + 2R + 2N) + O(2\,T \cdot EC \cdot N).
\label{eq:mac-ssm}
\end{equation}
Phần so khớp của nhánh cục bộ gồm tích $\mat{Q}\mat{K}^{\top}$ và tích với $\mat{V}$, mỗi tích tốn $h \cdot d = C$ MAC cho một cặp token. Khi tính đầy đủ ma trận $T \times T$ rồi che như cài đặt hiện tại, chi phí là $2T^2C$. Khi chỉ tính các cặp nằm trong dải, số cặp hợp lệ là
\begin{equation}
P(T, w) = T(2w + 1) - w(w + 1) \leq TW,
\label{eq:band-pairs}
\end{equation}
vì các hàng ở hai biên có ít hàng xóm hơn. Khi đó, chi phí giảm còn $2C \cdot P(T, w) \leq 2TWC$. Về bộ nhớ, ma trận trọng số attention của một chuỗi có $hT^2$ phần tử ở dạng đầy đủ và $h \cdot P(T, w)$ phần tử ở dạng dải còn các kích hoạt trung gian của các phép chiếu có kích thước $O(T \cdot EC)$. Trạng thái của SSM có $EC \cdot N$ phần tử cho mỗi chiều. Kernel CUDA của \code{mamba-ssm} giữ trạng thái này trong bộ nhớ trên chip và tính lại khi lan truyền ngược thay vì lưu toàn bộ tensor $T \times EC \times N$ \cite{gu2023mamba} nên bộ nhớ kích hoạt của nhánh SSM vẫn tuyến tính theo $T$.

Từ $2T^2C > 4TC^2$, so khớp toàn cục vượt chi phí các phép chiếu của chính nhánh attention khi $T > 2C$, tức $T > 1536$ ở tầng 4 và $T > 768$ ở tầng 3. Về bộ nhớ, tỉ lệ giữa ma trận attention đầy đủ và dạng dải là $T^2/P(T, w) \approx T/W$ nên tăng tuyến tính theo $T$. Như vậy, nhận định ``tuyến tính theo $T$'' ở \cref{tab:necks} mô tả cách chi phí \emph{tăng} khi đoạn video dài ra chứ không có nghĩa là ở $T = 16$ khối rẻ hơn attention toàn cục một cách đáng kể. Ở độ dài này, chi phí bị chi phối bởi các phép chiếu bậc $C^2$ mà mọi temporal neck dạng Transformer đều phải trả.

\section{Lý do lựa chọn các tham số thiết kế}
\label{sec:design-rationale}

Mục này trình bày lập luận thiết kế đằng sau các giá trị mặc định của module. Các giá trị này được chọn từ lập luận và từ giá trị mặc định của các cài đặt gốc, không được tinh chỉnh trên dữ liệu và không có ablation riêng cho từng tham số trong luận văn. Vì vậy, các lập luận dưới đây chỉ giải thích vì sao lựa chọn là hợp lý chứ không khẳng định đó là lựa chọn tối ưu.

\paragraph{Cửa sổ $W = 5$.} Với đoạn 32 khung hình lấy mẫu cách nhau 2 khung hình gốc và patch embedding thời gian có bước 2, mỗi token tại neck đại diện cho 2 khung hình lấy mẫu, tương ứng 4 khung hình gốc. Cửa sổ 5 token do đó phủ 10 khung hình lấy mẫu, khoảng một phần ba đoạn video, đủ để chứa một pha chuyển tiếp ngắn như đưa điện thoại lên nhưng vẫn nhỏ hơn hẳn độ dài đoạn ($W < T = 16$) nên nhánh cục bộ thực sự cục bộ và không trùng vai trò với nhánh SSM. Kích thước lẻ giúp cửa sổ đối xứng quanh mỗi token ($w = 2$ về mỗi phía), phù hợp với việc đoạn video được lấy đối xứng quanh khung hình khóa. Khi $W$ tăng tới mức phủ toàn đoạn, nhánh này trở thành attention toàn cục như cấu hình A1.

\paragraph{Hai khối.} Khi xếp chồng hai khối, trường tiếp nhận của riêng nhánh cục bộ mở rộng từ $2w + 1 = 5$ lên $4w + 1 = 9$ token và cổng có hai lần định tuyến, lần sau dựa trên đặc trưng đã được trộn ở lần trước. Mặt khác, theo \cref{eq:params-block}, mỗi khối ở tầng 4 thêm gần 7 triệu tham số nên hai khối đã chiếm khoảng một nửa số tham số của backbone. Với lượng dữ liệu nhỏ của bài toán phân xưởng, giữ neck nhỏ hơn backbone là lựa chọn thận trọng để hạn chế quá khớp.

\paragraph{Bốn đầu attention.} Do $C$ phải chia hết cho $h$ nên $h = 4$ cho số chiều mỗi đầu $d = 192$ ở tầng 4 và $d = 96$ ở tầng 3, tức là mỗi đầu vẫn đủ rộng ở tầng có ít kênh hơn trong khi mô hình có nhiều không gian con học các kiểu chuyển động khác nhau. Theo \cref{eq:mac-lin}, số đầu không làm thay đổi chi phí phép chiếu và chi phí so khớp $2T^2C$ cũng không phụ thuộc $h$. Số đầu chỉ ảnh hưởng tới kích thước ma trận trọng số attention $hT^2$, vốn rất nhỏ ở $T = 16$.

\paragraph{Kích thước trạng thái $N = 16$, $E = 2$ và $k = 4$.} Đây là các giá trị mặc định của khối Mamba gốc \cite{gu2023mamba} và của cài đặt \code{bimamba v2} \cite{chen2024videomambasuite}. Giữ nguyên các giá trị này giúp dùng được kernel CUDA đã được kiểm chứng rộng rãi và tránh đưa thêm siêu tham số cần tinh chỉnh. Theo \cref{eq:mac-ssm}, chi phí quét tăng tuyến tính theo $N$ nhưng ở thiết kế hiện tại vẫn nhỏ hơn nhiều so với phần phép chiếu (\cref{tab:flops}).

\paragraph{DropPath $p = 0{,}1$.} DropPath là một cách điều chuẩn phù hợp với khối residual vì khi nhánh dư của một chuỗi bị loại bỏ, đầu ra của khối đúng bằng đầu vào và pipeline vẫn hoạt động như khi không có neck. Điều này khuyến khích neck chỉ học phần bổ sung thay vì thay thế đặc trưng của backbone. Neck dùng cùng một tỉ lệ $0{,}1$ cho cả hai khối, thấp hơn giá trị \code{drop\_path\_rate} $= 0{,}2$ đặt cho backbone trong cấu hình, vì neck chỉ có hai khối và việc loại bỏ quá thường xuyên sẽ làm giảm tín hiệu học của chính các nhánh mới.

\paragraph{Chuẩn hóa trước.} Với pre-norm, đường residual từ $\mat{X}$ tới $\mat{Y}$ trong \cref{eq:residual} là phép đồng nhất không đi qua chuẩn hóa nào nên gradient truyền thẳng về backbone và thống kê của đặc trưng tiền huấn luyện được giữ nguyên khi các nhánh còn đóng góp nhỏ. Nếu dùng post-norm, LayerNorm sẽ nằm trên đường chính và làm thay đổi thang đo của đặc trưng đưa vào RoIAlign ngay từ đầu huấn luyện. Pre-norm cũng là cách bố trí thường cho huấn luyện ổn định hơn với các khối dạng Transformer \cite{xiong2020prenorm}.

\section{Thành phần tích hợp 1: đặc trưng đa tỉ lệ}
\label{sec:multiscale}

Thành phần này xử lý hạn chế L3. Video Swin-T tạo ra bốn tầng đặc trưng với $96, 192, 384, 768$ kênh và bước giảm mẫu lần lượt là $4, 8, 16, 32$. Nếu chỉ dùng tầng 4 thì những vật nhỏ như điện thoại hay bàn chân chiếm chưa đến một ô trên lưới $8 \times 8$. Vì vậy, pipeline đầy đủ đặt \code{out\_indices = (2, 3)} để lấy thêm tầng 3 và tạo \emph{một ngăn xếp Hybrid FGTN-Mamba riêng cho mỗi tầng} (\code{in\_channels = (384, 768)}) mà không chia sẻ tham số. Sau neck, \code{SingleRoIExtractor3D} nội suy bản đồ tầng 4 về độ phân giải của tầng 3, gộp trung bình theo thời gian, nối theo kênh rồi cắt RoI với tỷ lệ $1/16$:
\begin{equation}
r_k = \RoIAlign\Big(\Concat\big[\Pool_T(\tilde{F}^{(3)}),\ \Pool_T(\uparrow_2 \tilde{F}^{(4)})\big],\ b_k\Big) \in \R^{1152 \times 1 \times 8 \times 8}.
\label{eq:multiscale}
\end{equation}
Phép gộp theo thời gian được cố ý thực hiện \emph{sau} khi mỗi tầng đã nhận ngữ cảnh dài--ngắn từ neck. Tầng 3 có số chuỗi thời gian gấp bốn lần tầng 4 nên kích thước lô của cấu hình đa tỉ lệ cần được chọn theo bộ nhớ khả dụng. Đóng góp của thành phần này được tách riêng qua cặp cấu hình A6 $\rightarrow$ A7.

\section{Thành phần tích hợp 2: mạng quan hệ actor--cảnh (ACRN)}
\label{sec:acrn}

ACRN \cite{sun2018acrn} xử lý hạn chế L4, xuất phát từ thực tế là trong phân xưởng, cùng một tư thế có thể mang ý nghĩa khác nhau tùy theo bàn, ghế và thiết bị xung quanh người. Luận văn dùng \code{ACRNHead} của MMAction2 làm \code{shared\_head} trong \code{AVARoIHead}. Với đặc trưng RoI $r_k$ và bản đồ cảnh $G$ (đã gộp theo thời gian, $1152 \times 1 \times h \times w$), đầu ra của ACRN là
\begin{multline}
c_k = \mathrm{Conv}_{1\times3\times3}\Big(\mathrm{Conv}_{1\times1\times1}\big(\Concat[\Tile_{h \times w}(\max\!\Pool(r_k)),\ G_{\text{batch}(k)}]\big)\Big)\\
\in \R^{1152 \times 1 \times h \times w},
\label{eq:acrn}
\end{multline}
trong đó mỗi tích chập đều kèm BatchNorm và ReLU. Vectơ actor được lát (tile) lên toàn bộ lưới cảnh nên quan hệ được điều kiện hóa theo từng actor, nhờ đó hai người trong cùng một khung hình sẽ nhận ngữ cảnh khác nhau. So với gộp trung bình toàn cục, ACRN giữ được bố cục không gian của cảnh. So với STAR \cite{gritsenko2024star}, ACRN vẫn phụ thuộc vào proposal nhưng tích hợp được trực tiếp vào pipeline hai giai đoạn.

\section{Thành phần tích hợp 3: hàm mất mát hai cấp và ngưỡng theo lớp}
\label{sec:loss}

Head \code{BBoxHeadAVA} của MMAction2 chỉ giữ lại các RoI có ít nhất một nhãn dương ($\mathcal{P}$) và tính BCE trên $K$ logit sigmoid:
\begin{equation}
\mathcal{L}_{\text{BCE}} = \frac{1}{|\mathcal{P}| \cdot K} \sum_{k \in \mathcal{P}} \sum_{c=1}^{K}
  \BCE\big(s_{k,c}, y_{k,c}\big).
\label{eq:bce}
\end{equation}
Để xử lý hạn chế L5, head \code{BBoxHeadAVAWeighted} giữ nguyên phần chọn RoI và chỉ thay biểu thức mất mát bằng focal loss có trọng số lớp,
\begin{equation}
\begin{gathered}
\mathcal{L} = \frac{1}{|\mathcal{P}| \cdot K} \sum_{k \in \mathcal{P}} \sum_{c=1}^{K}
  w_c\, \alpha\, (1 - p_{t,k,c})^{\gamma}\, \BCE(s_{k,c}, y_{k,c}),\\
p_{t,k,c} = \exp\big(-\BCE(s_{k,c}, y_{k,c})\big),
\end{gathered}
\label{eq:focal}
\end{equation}
với $\gamma = 2$ và $\alpha = 1$. Khi $\gamma = 0$ và $w_c = 1$, \cref{eq:focal} trùng với \cref{eq:bce}. Trọng số lớp được tính theo số mẫu hiệu dụng \cite{cui2019classbalanced}:
\begin{equation}
w_c = \frac{\tilde{w}_c}{\frac{1}{K}\sum_{c'} \tilde{w}_{c'}}, \qquad
\tilde{w}_c = \frac{1 - \beta}{1 - \beta^{\max(n_c, 1)}}, \qquad \beta = 0{,}999,
\label{eq:classweight}
\end{equation}
sau đó được chuẩn hóa để có trung bình bằng 1 và cắt trên tại 10. Sau khi huấn luyện, ngưỡng $\tau_c$ của từng lớp được chọn trên tập kiểm định theo tiêu chí nghiệp vụ (ví dụ cực đại F1 hoặc đạt recall tối thiểu). Do mAP không phụ thuộc ngưỡng nên độ đo này không bị ảnh hưởng bởi lựa chọn trên.

\section{Luồng suy luận và logic cảnh báo}
\label{sec:inference}

\begin{enumerate}
  \item lấy đoạn 32 khung hình (bước 2) đối xứng quanh khung hình khóa, đổi kích thước cạnh ngắn về 256 rồi đọc person proposal (hoặc chạy bộ phát hiện người);
  \item trích đặc trưng tầng 3 và tầng 4, chạy Hybrid FGTN-Mamba riêng trên từng tầng rồi lần lượt qua 3D RoIAlign đa tỉ lệ, ACRN và head sigmoid để thu được $p_{k,c}$;
  \item so sánh $p_{k,c}$ với $\tau_c$ và chỉ phát sự kiện khi cùng một actor vượt ngưỡng $m$ trong $n$ khung hình khóa liên tiếp nhằm giảm cảnh báo chớp tắt;
  \item ghép sự kiện theo định danh track, lưu thời điểm, hộp, lớp, điểm và ảnh minh chứng, đồng thời áp dụng thời gian chờ (cooldown) để tránh lặp cảnh báo.
\end{enumerate}
Các tham số $m/n$, cooldown và $\tau_c$ thuộc lớp nghiệp vụ và không được tối ưu trên tập kiểm tra. Độ trễ đầu--cuối bao gồm thời gian giải mã, bộ phát hiện người, bộ đệm nhìn trước của quét hai chiều (tối thiểu nửa đoạn video), mô hình và gửi cảnh báo nên thời gian suy luận của riêng mô hình không đại diện cho toàn hệ thống. Một cài đặt minh họa của luồng này trên video được trình bày ở \cref{sec:demo}. \Cref{tab:shapes} tóm tắt kích thước tensor qua pipeline.

\Cref{alg:pipeline} mô tả hình thức bước 2 cho một đoạn video. Các dòng trước vòng lặp chạy một lần cho mỗi đoạn còn các dòng trong vòng lặp chạy cho từng proposal. Trong cài đặt, vòng lặp này không chạy tuần tự mà được thực hiện theo lô trên mọi RoI của lô dữ liệu. Bản đồ cảnh mà ACRN dùng cho mỗi proposal là bản đồ của chính đoạn video chứa proposal đó. Biến thể chỉ dùng tầng 4 bỏ nhánh tầng 3 và ACRN, cắt RoI trực tiếp trên $\Pool_T(\tilde{F}^{(4)})$ với tỷ lệ $1/32$ rồi đưa $r_k \in \R^{768 \times 1 \times 8 \times 8}$ thẳng vào head.

\begin{algorithm}[htbp]
\caption{Lan truyền thuận của pipeline đầy đủ cho một đoạn video}
\label{alg:pipeline}
\begin{algorithmic}[1]
\Require đoạn video $V \in \R^{3 \times 32 \times 256 \times 256}$; các person proposal $\{b_k\}_{k=1}^{N}$ tại khung hình khóa
\Ensure xác suất $\mat{p}_k \in [0,1]^{K}$ cho từng proposal
\State $(F^{(3)}, F^{(4)}) \gets \text{Swin-T}(V)$ với \code{out\_indices} $= (2, 3)$
\For{$s \in \{3, 4\}$}
  \State $\tilde{F}^{(s)} \gets \text{HFM}_s(F^{(s)})$ theo \cref{alg:block} với tham số riêng của tầng $s$
\EndFor
\State $\hat{F} \gets \Concat\big[\Pool_T(\tilde{F}^{(3)}),\ \Pool_T(\uparrow_2 \tilde{F}^{(4)})\big] \in \R^{1152 \times 1 \times 16 \times 16}$
\For{$k = 1, \dots, N$}
  \State $r_k \gets \RoIAlign(\hat{F}, b_k)$ với tỷ lệ $1/16$ và lưới $8 \times 8$ \Comment{\cref{eq:multiscale}}
  \State $c_k \gets \mathrm{ACRN}(r_k, \hat{F}) \in \R^{1152 \times 1 \times 16 \times 16}$ \Comment{\cref{eq:acrn}}
  \State $\mat{v}_k \gets$ gộp $c_k$ theo thời gian và không gian thành vectơ $1152$ chiều
  \State $\mat{s}_k \gets \mat{W}_{\text{cls}}\mat{v}_k + \mat{b}_{\text{cls}} \in \R^{K+1}$
  \State $\mat{p}_k \gets \sigma(\mat{s}_k)$ trên $K$ cột hành vi, bỏ cột nền
\EndFor
\State \Return $\{\mat{p}_k\}_{k=1}^{N}$
\end{algorithmic}
\end{algorithm}

Diễn giải theo từng bước trong \cref{tab:shapes}, patch embedding với nhân $(2, 4, 4)$ giảm trục thời gian từ 32 xuống 16 và lưới không gian từ $256 \times 256$ xuống $64 \times 64$, đồng thời chiếu lên 96 kênh. Ba lần gộp patch (patch merging) sau đó mỗi lần giảm một nửa chiều cao, chiều rộng và tăng gấp đôi số kênh nhưng không động tới trục thời gian nên $T = 16$ được giữ nguyên tới tầng 4, với tầng 3 có $384$ kênh trên lưới $16 \times 16$ và tầng 4 có $768$ kênh trên lưới $8 \times 8$. Neck định dạng lại mỗi tầng thành $B \cdot 256$ hoặc $B \cdot 64$ chuỗi độ dài 16 theo \cref{eq:reshape} rồi trả về đúng kích thước ban đầu nên bộ trích RoI không cần biết có neck hay không. Tại bộ trích RoI, bản đồ tầng 4 được nội suy về lưới $16 \times 16$ của tầng 3, cả hai được gộp trung bình theo thời gian và nối theo kênh thành $384 + 768 = 1152$ kênh. Bản đồ này vừa là nguồn để cắt RoI $8 \times 8$ cho từng proposal vừa là bản đồ cảnh của ACRN. Trong ACRN, đặc trưng RoI được gộp cực đại thành một vectơ $1152$ chiều, lát lên lưới $16 \times 16$ và nối với bản đồ cảnh thành $2304$ kênh. Tích chập $1 \times 1 \times 1$ đưa số kênh về $1152$ còn tích chập $1 \times 3 \times 3$ giữ nguyên lưới $16 \times 16$. Cuối cùng, head gộp đặc trưng này thành một vectơ, áp dụng dropout với tỉ lệ $0{,}5$ khi huấn luyện và một lớp tuyến tính cho $K + 1$ logit, trong đó cột nền không được dùng trong hàm mất mát.

\Cref{alg:alert} mô tả hình thức bước 3 và bước 4 cho từng khung hình khóa. Mỗi cặp track--lớp giữ một bộ đệm $n$ kết quả so ngưỡng gần nhất và một mốc thời gian cảnh báo gần nhất. Sự kiện chỉ được phát khi ít nhất $m$ trong $n$ khung hình khóa liên tiếp của cùng một track vượt ngưỡng $\tau_c$ và khoảng cách tới lần cảnh báo trước của cùng cặp track--lớp không nhỏ hơn thời gian chờ $\Delta_{\text{cd}}$. Với $m > 1$, một điểm số vọt lên ở một khung hình khóa đơn lẻ không đủ để phát cảnh báo nhưng đổi lại, sự kiện được phát chậm hơn khung hình khóa vượt ngưỡng đầu tiên ít nhất $m - 1$ khoảng cách giữa các khung hình khóa, cộng thêm độ trễ nhìn trước của quét hai chiều. Thuật toán chỉ nêu cấu trúc của logic. Các giá trị $\tau_c$, $m$, $n$ và $\Delta_{\text{cd}}$ là tham số nghiệp vụ, được đặt theo yêu cầu vận hành của từng phân xưởng và không được tinh chỉnh trong luận văn.

\begin{algorithm}[htbp]
\caption{Logic cảnh báo theo track tại một khung hình khóa}
\label{alg:alert}
\begin{algorithmic}[1]
\Require thời điểm khung hình khóa $t$; tập track $\mathcal{A}_t$, mỗi track $a$ có hộp $b_a$ và xác suất $p_{a,c}$ từ \cref{alg:pipeline}; ngưỡng $\tau_c$; tham số $m \leq n$; thời gian chờ $\Delta_{\text{cd}}$; bộ đệm $\mathcal{H}_{a,c}$ chứa tối đa $n$ kết quả gần nhất; mốc cảnh báo gần nhất $t^{\text{last}}_{a,c}$ (khởi tạo $-\infty$)
\Ensure tập sự kiện $\mathcal{E}_t$
\State $\mathcal{E}_t \gets \emptyset$
\For{mỗi track $a \in \mathcal{A}_t$ và mỗi lớp $c$}
  \State $v \gets 1$ nếu $p_{a,c} \geq \tau_c$; ngược lại $v \gets 0$
  \State thêm $v$ vào $\mathcal{H}_{a,c}$ và bỏ phần tử cũ nhất nếu $\mathcal{H}_{a,c}$ có hơn $n$ phần tử
  \If{$\sum_{v' \in \mathcal{H}_{a,c}} v' \geq m$ và $t - t^{\text{last}}_{a,c} \geq \Delta_{\text{cd}}$}
    \State thêm $(t, a, b_a, c, p_{a,c}, \text{ảnh minh chứng})$ vào $\mathcal{E}_t$
    \State $t^{\text{last}}_{a,c} \gets t$
  \EndIf
\EndFor
\State \Return $\mathcal{E}_t$
\end{algorithmic}
\end{algorithm}

\section{Rủi ro kỹ thuật và cách giám sát}
\label{sec:risks}

\begin{itemize}
  \item \emph{cổng bão hòa}, khi $g$ dồn về một nhánh, được giám sát bằng cách theo dõi phân bố của $g$ và chuẩn gradient của từng nhánh theo epoch;
  \item \emph{tầng lấn át}, khi tầng 4 lấn át tầng 3 trong cấu hình đa tỉ lệ, được giám sát bằng cách đo chuẩn đặc trưng và ablation từng tầng;
  \item \emph{điểm nóng bộ nhớ} xuất hiện do tầng 3 tạo ra $B \cdot 256$ chuỗi nên cần theo dõi bộ nhớ đỉnh theo kích thước lô, $T$ và độ phân giải;
  \item \emph{độ trễ hai chiều} xuất hiện do quét ngược cần các khung hình tương lai nên cần báo cáo cả thông lượng lẫn độ trễ đầu--cuối;
  \item \emph{sai lệch tỷ lệ RoI} được tránh bằng cách đặt \code{featmap\_stride} của bộ trích RoI khớp với bước giảm mẫu của đặc trưng (32 với tầng 4, 16 với cấu hình đa tỉ lệ);
  \item \emph{bão hòa độ đo} xảy ra khi dữ liệu nhỏ khiến mAP cao có thể che khuất lỗi nên cần kiểm tra thêm theo camera (cách chia S2, \cref{sec:splits}), trên mẫu âm khó và sai số hiệu chỉnh.
\end{itemize}

\section*{Kết luận chương 3}
\addcontentsline{toc}{section}{Kết luận chương 3}

Chương này đã mô tả Hybrid FGTN-Mamba gồm nhánh Bi-SSM cho phụ thuộc dài hạn, nhánh W-MHSA cho chuyển tiếp ngắn hạn và cổng sigmoid định tuyến theo kênh và bước thời gian, với chi phí tuyến tính theo độ dài chuỗi. Khối được đặt tại vị trí hậu-backbone, tiền-RoI và được áp dụng độc lập trên hai tầng đặc trưng. Pipeline đầy đủ còn bổ sung multi-scale RoI, ACRN, hàm mất mát hai cấp và logic cảnh báo theo track. Các chế độ nhánh của cùng một module cho phép hiện thực phần lớn ma trận ablation trình bày trong \cref{ch:experiments}.
